def create_dataloaders(size=224, bs=64, augment=True):
    """Create dataloaders with progressive resizing capability"""
    
    # Item transforms - resize and normalize
    item_tfms = [Resize(size, method='squish')]
    
    # Batch transforms - augmentation
    if augment:
        batch_tfms = [
            *aug_transforms(
                size=size, 
                min_scale=0.75,
                do_flip=True,
                flip_vert=True,  # Important for leaf images
                max_rotate=30,
                max_zoom=1.2,
                max_lighting=0.3,
                max_warp=0.15
            ),
            Normalize.from_stats(*imagenet_stats)
        ]
    else:
        batch_tfms = [Normalize.from_stats(*imagenet_stats)]
    
    return ImageDataLoaders.from_folder(
        train_path,
        valid_pct=0.2,
        seed=42,
        item_tfms=item_tfms,
        batch_tfms=batch_tfms,
        bs=bs
    )

def train_model(arch, size=224, epochs=10, mixup_alpha=0.2, tta_n=5, save_name=None):
    """Train a single model with all optimizations"""
    
    print(f"\n{'='*50}")
    print(f"Training {arch}")
    print(f"Image size: {size}, Epochs: {epochs}")
    print(f"{'='*50}")
    
    # Create dataloaders
    dls = create_dataloaders(size=size)
    
    # Callbacks
    cbs = []
    if mixup_alpha > 0:
        cbs.append(MixUp(alpha=mixup_alpha))  # Changed: added 'alpha=' parameter name
        print(f"Using MixUp with alpha={mixup_alpha}")
    
    # Create model
    model = timm.create_model(arch, pretrained=True, num_classes=dls.c)
    model = model.to(memory_format=torch.channels_last)
    learn = Learner(dls, model, metrics=[accuracy, top_k_accuracy], cbs=cbs).to_fp16()
    
    # Find learning rate
    lr = learn.lr_find(suggest_funcs=(valley, slide)).valley
    print(f"Suggested learning rate: {lr}")
    
    # Train with cosine annealing for better results
    learn.fit_one_cycle(epochs, lr_max=lr, div=25, pct_start=0.3)  # Modified
    
    # Get validation results
    val_loss, val_acc, val_top5 = learn.validate()
    
    # TTA predictions - reduced from 5 to 3 for speed
    print(f"\nPerforming TTA with {tta_n} augmentations...")
    tta_preds, _ = learn.tta(n=3)  # Reduced from 5 to 3
    tta_acc = accuracy(tta_preds, learn.get_preds(dl=learn.dls.valid)[1])
    
    results = {
        'model': arch,
        'size': size,
        'epochs': epochs,
        'val_loss': val_loss,
        'val_acc': float(val_acc),
        'val_top5': float(val_top5),
        'tta_acc': float(tta_acc),
        'learn': learn
    }
    
    # Save model if specified
    if save_name:
        learn.save(save_name)
        print(f"Model saved as {save_name}")
    
    return results

# Train with progressive resizing for best model
def progressive_training(arch='convnext_small.fb_in22k', final_size=320):
    """Implement progressive resizing strategy from notebook 9"""
    
    print(f"\nProgressive training for {arch}")
    print("="*60)
    
    # Stage 1: Small images, quick training
    print("\nStage 1: 128x128 images")
    dls = create_dataloaders(size=128, bs=128)
    model = timm.create_model(arch, pretrained=True, num_classes=dls.c)
    learn = Learner(dls, model, metrics=[accuracy]).to_fp16()
    learn.fit_one_cycle(3, 3e-3)
    
    # Stage 2: Medium images
    print("\nStage 2: 224x224 images")
    learn.dls = create_dataloaders(size=224, bs=64)
    learn.fine_tune(5, 1e-3)
    
    # Stage 3: Large images
    print(f"\nStage 3: {final_size}x{final_size} images")
    learn.dls = create_dataloaders(size=final_size, bs=32)
    learn.fine_tune(5, 1e-4)
    
    # Final TTA evaluation
    tta_preds, _ = learn.tta(n=5)
    tta_acc = accuracy(tta_preds, learn.get_preds(dl=learn.dls.valid)[1])
    print(f"\nFinal TTA accuracy: {tta_acc.item():.4f}")
    
    return learn

def create_ensemble(model_results, test_dl):
    """Create ensemble predictions from multiple models"""
    
    print("\nCreating ensemble predictions...")
    
    # Filter out poor performers (exclude models with TTA acc < 0.95)
    good_models = [r for r in model_results if r['tta_acc'] > 0.95]
    print(f"Using {len(good_models)} models out of {len(model_results)} total")
    
    all_preds = []
    weights = []
    
    for result in good_models:
        learn = result['learn']
        # Get TTA predictions for test set - reduced to 3 augmentations
        preds, _ = learn.tta(dl=test_dl, n=3)  # Reduced from 5
        all_preds.append(preds)
        weights.append(result['tta_acc'])  # Weight by accuracy
        print(f"Added {result['model']} (acc: {result['tta_acc']:.4f}) to ensemble")
    
    # Weighted average based on TTA accuracy
    weights = torch.tensor(weights)
    weights = weights / weights.sum()
    
    ensemble_preds = sum(pred * w for pred, w in zip(all_preds, weights))
    return ensemble_preds

def create_submission(preds, filename='submission.csv'):
    """Create Kaggle submission file"""
    
    # Get predicted classes
    idxs = preds.argmax(dim=1)
    
    # Map to class names
    vocab = learn.dls.vocab
    labels = [vocab[idx] for idx in idxs]
    
    # Create submission dataframe
    ss = pd.read_csv(path/'sample_submission.csv')
    ss['label'] = labels
    ss.to_csv(filename, index=False)
    
    print(f"Submission saved to {filename}")
    print(f"First 10 predictions:")
    print(ss.head(10))
    
    return ss
